   169→    function initAudio() {
   170→      if (!audioCtx) {

   172→        masterGain = audioCtx.createGain();
   173→        masterGain.gain.value = (state.settings.masterVolume || 50) / 100;

   179→    function createNoiseBuffer(type = 'white', duration = 2) {
   180→      const sampleRate = audioCtx.sampleRate;
   181→      const length = sampleRate * duration;
   182→      const buffer = audioCtx.createBuffer(1, length, sampleRate);
   183→      const data = buffer.getChannelData(0);
   184→      let lastOut = 0;
   185→      for (let i = 0; i < length; i++) {
   186→        const white = Math.random() * 2 - 1;
   187→        if (type === 'white') data[i] = white;
   188→        else if (type === 'pink' || type === 'brown') {
   189→          data[i] = (lastOut + (0.02 * white)) / 1.02;
   190→          lastOut = data[i];
   191→          data[i] *= 3.5;
   192→        }
   193→      }
   194→      return buffer;
   195→    }

   197→    function playCrystalChime() {
   198→      if (state.settings.batterySaver) return;
   199→      initAudio();
   200→      const frequencies = [1318.51, 1567.98, 2093.00, 2637.02];
   201→      const now = audioCtx.currentTime;
   202→      const gain = audioCtx.createGain();
   203→      gain.gain.setValueAtTime(0, now);
   204→      gain.gain.linearRampToValueAtTime(0.15, now + 0.02);
   205→      gain.gain.exponentialRampToValueAtTime(0.001, now + 1.5);
   206→      gain.connect(masterGain);
   207→      frequencies.forEach((freq, i) => {
   208→        const osc = audioCtx.createOscillator();
   209→        const oscGain = audioCtx.createGain();
   210→        osc.type = 'sine';
   211→        osc.frequency.value = freq;
   212→        oscGain.gain.setValueAtTime(0, now);
   213→        oscGain.gain.linearRampToValueAtTime(0.3 / (i + 1), now + 0.01 + i * 0.05);
   214→        oscGain.gain.exponentialRampToValueAtTime(0.001, now + 1.5);
   215→        osc.connect(oscGain);
   216→        oscGain.connect(gain);
   217→        osc.start(now + i * 0.05);
   218→        osc.stop(now + 1.5);
   219→      });

   765→    function isoWeekKey(d) {
   766→      const date = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
   767→      const dayNum = date.getUTCDay() || 7;
   768→      date.setUTCDate(date.getUTCDate() + 4 - dayNum);
   769→      const yearStart = new Date(Date.UTC(date.getUTCFullYear(), 0, 1));
   770→      const weekNo = Math.ceil((((date - yearStart) / 86400000) + 1) / 7);
   771→      return date.getUTCFullYear() + '-W' + weekNo;
   772→    }

   774→    function computeWeeklyStats() {
   775→      const days = [];
   776→      for (let i = 6; i >= 0; i--) {
   777→        const d = new Date(); d.setDate(d.getDate() - i);
   778→        days.push(d.toISOString().split('T')[0]);
   779→      }
   780→      const dayNames = ['یکشنبه', 'دوشنبه', 'سه‌شنبه', 'چهارشنبه', 'پنجشنبه', 'جمعه', 'شنبه'];
   781→      const perDay = days.map(ds => {
   782→        const d = new Date(ds + 'T00:00:00');
   783→        return { date: ds, dayFa: dayNames[d.getDay()], total: state.habits.length, done: state.habits.filter(h => h.completions[ds]).length };
   784→      });
   785→      const totalSlots = perDay.reduce((a, d) => a + d.total, 0);
   786→      const totalDone = perDay.reduce((a, d) => a + d.done, 0);
   787→      const rate7 = totalSlots > 0 ? Math.round((totalDone / totalSlots) * 100) : 0;
   788→      const bestStreak = state.habits.length ? Math.max(...state.habits.map(h => h.streak || 0)) : 0;
   789→      const moodScores = (state.moodLog || []).filter(m => days.includes(m.date)).map(m => m.score);
   790→      const moodAvg = moodScores.length ? Math.round((moodScores.reduce((a, b) => a + b, 0) / moodScores.length) * 10) / 10 : null;
   791→      const ranked = perDay.filter(d => d.total > 0).slice().sort((a, b) => (b.done / b.total) - (a.done / a.total));
   792→      const habitRows = state.habits.map(h => ({ name: h.name, category: h.category, streak: h.streak || 0, done7: days.filter(ds => h.completions[ds]).length }));
   793→      habitRows.sort((a, b) => b.done7 - a.done7);
   794→      return {
   795→        rate7: rate7,
   796→        totalDone: totalDone,
   797→        totalSlots: totalSlots,
   798→        totalHabits: state.habits.length,
   799→        bestStreak: bestStreak,
   800→        moodAvg: moodAvg,
   801→        moodCount: moodScores.length,
   802→        bestDay: ranked.length ? ranked[0].dayFa : null,
   803→        weakDay: ranked.length > 1 ? ranked[ranked.length - 1].dayFa : null,
   804→        topHabits: habitRows.slice(0, 3),
   805→        zeroHabits: habitRows.filter(h => h.done7 === 0).map(h => h.name).slice(0, 5),
   806→        perDay: perDay
   807→      };
   808→    }
